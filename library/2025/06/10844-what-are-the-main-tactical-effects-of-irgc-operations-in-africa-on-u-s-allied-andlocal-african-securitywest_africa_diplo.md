```
}

// --- Main SVG setup
width = 700;
height = 400;
svg = select(DOM.svg(width, height));

// --- Force simulation
simulation = forceSimulation(nodes)
  .force("link", forceLink(links).id(d=>d.id).distance(180))
  .force("charge", forceManyBody().strength(-220))
  .force("center", forceCenter(width/2, height/2))
  .stop();

// Run simulation a bit
for (let i=0; i<180; ++i) simulation.tick();

// --- Render links (escalation = thickness, color)
svg.append("g").selectAll("line").data(links)
  .join("line")
  .attr("x1", d=>d.source.x)
  .attr("y1", d=>d.source.y)
  .attr("x2", d=>d.target.x)
  .attr("y2", d=>d.target.y)
  .attr("stroke-width", d=>2 + 8*d.escalation)
  .attr("stroke", d=>`rgba(255,80,0,${d.escalation})`);

// --- Render nodes, pulse according to risk + month
svg.append("g").selectAll("circle").data(nodes)
  .join("circle")
  .attr("cx", d=>d.x)
  .attr("cy", d=>d.y)
  .attr("r", d=>25+25*modRisk(d.risk, month))
  .attr("fill", d => riskColor(modRisk(d.risk, month)))
  .attr("stroke", "#222")
  .attr("stroke-width", 2);

// --- Node labels
svg.append("g").selectAll("text").data(nodes)
  .join("text")
  .attr("x", d=>d.x)
  .attr("y", d=>d.y-40)
  .attr("text-anchor", "middle")
  .attr("font-size", 16)
  .attr("fill", "#111")
  .text(d=>d.id);

// --- Add timeline label
svg.append("text")
  .attr("x", width-160)
  .attr("y", 42)
  .attr("font-size", 18)
  .attr("fill","#333")
  .text(["June","July","August","September"][month-6] + " 2025")

svg.node()
```
