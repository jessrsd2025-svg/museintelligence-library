### 2. FastAPI app with Autosummarization and Section Confidence


```
import os
import openai
from fastapi import FastAPI, UploadFile, File, Form
from typing import List, Dict, Optional
from PyPDF2 import PdfReader

openai.api_key = os.getenv("OPENAI_API_KEY")
app = FastAPI()

SYSTEM_PROMPT = """
You are an expert document analyst and auto-summarizer.
Given a long text (such as a US Congressional report), do the following:
1. Identify all major sections and their likely intended section/titles.
2. Create a concise summary of each section (2-5 sentences max).
3. For each section, return a confidence score (1.0=very certain, 0.0=no confidence) that the detected section title accurately reflects the content.
Return a markdown bullet list with: "section_title", "summary", and "confidence_score".
"""

def extract_pdf_text(pdf_bytes: bytes, maxpages: Optional[int]=None) -> str:
    reader = PdfReader(io.BytesIO(pdf_bytes))
    pages = reader.pages if maxpages is None else reader.pages[:maxpages]
    return "\n\n".join(page.extract_text() for page in pages if page.extract_text())

def gpt_section_summarize(text: str, model: str = "gpt-3.5-turbo") -> List[Dict]:
    chat = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "user", "content": text[:12000]}  # truncate for cheap models, or iterate per chunk
    ]
    res = openai.ChatCompletion.create(
        model=model,
        messages=chat,
        temperature=0.1
    )
    # Output markdown-parsed (optional: use regex for extracting fields)
    return res['choices'][0]['message']['content']

@app.post("/autosummarize-pdf/")
async def autosummarize_pdf(file: UploadFile = File(...), maxpages: int = Form(10)):
    # Read PDF, extract first N pages for speed
    content = await file.read()
    pdf_text = extract_pdf_text(content, maxpages=maxpages)
    summary_md = gpt_section_summarize(pdf_text)
    return {"ok": True, "summaries_markdown": summary_md}

@app.post("/autosummarize-text/")
async def autosummarize_text(payload: Dict):
    text = payload['text']
    summary_md = gpt_section_summarize(text)
    return {"ok": True, "summaries_markdown": summary_md}
```

import os
import io
import asyncio
import openai
from fastapi import FastAPI, UploadFile, File
from typing import List, Dict, Any, Optional
from PyPDF2 import PdfReader

from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.chat_models import ChatOpenAI
# For Azure, use: from langchain.chat_models import AzureChatOpenAI

#####################
# CONFIGURATION
#####################

USE_AZURE = bool(os.environ.get("OPENAI_API_TYPE") == "azure")  # Set this ENV for Azure
OPENAI_API_KEY = os.environ["OPENAI_API_KEY"]
OPENAI_API_BASE = os.environ.get("OPENAI_API_BASE")  # For Azure
OPENAI_API_VERSION = os.environ.get("OPENAI_API_VERSION", "2023-05-15")  # for Azure

MODEL = "gpt-3.5-turbo"
CHUNK_SIZE = 4000

if USE_AZURE:
    openai.api_type = "azure"
    openai.api_base = OPENAI_API_BASE
    openai.api_version = OPENAI_API_VERSION
    # When using Azure, you must set your model deployment name as the model param

##########################
# UTILITY FUNCTIONS
##########################

def extract_pdf_text(pdf_bytes: bytes, maxpages: Optional[int] = None) -> List[str]:
    """Return a list of text per page."""
    reader = PdfReader(io.BytesIO(pdf_bytes))
    pages = reader.pages if maxpages is None else reader.pages[:maxpages]
    return [page.extract_text() or "" for page in pages]

def chunk_with_langchain(texts: List[str], chunk_size: int = CHUNK_SIZE) -> List[str]:
    combined = "\n\n".join(texts)
    splitter = RecursiveCharacterTextSplitter(chunk_size=chunk_size, chunk_overlap=120)
    chunks = splitter.split_text(combined)
    return [chunk.strip() for chunk in chunks if chunk.strip()]

async def async_gpt_summarize(chunk: str, model: str = MODEL) -> str:
    if USE_AZURE:
        llm = ChatOpenAI(
            openai_api_key=OPENAI_API_KEY,
            openai_api_base=OPENAI_API_BASE,
            openai_api_version=OPENAI_API_VERSION,
            openai_api_type="azure",
            deployment_name=model,
            temperature=0.15,
            max_tokens=512,
        )
    else:
        llm = ChatOpenAI(
            openai_api_key=OPENAI_API_KEY,
            temperature=0.15,
            model=model,
            max_tokens=512,
        )
    prompt = (
        "Summarize the following section, give the best matching section title as a guess, "
        "and provide a confidence score (float 0.0-1.0) for your section title guess. "
        "Return as JSON: {\"section_title\": ..., \"summary\": ..., \"confidence\": ...}\n\n"
        f"Section:\n{text_shortener(chunk, 3800)}"
    )
    resp = await llm.apredict(prompt)
    return resp

def text_shortener(text: str, max_tokens: int) -> str:
    """Cut text to max_tokens if necessary (by character length proxy)."""
    return text[:max_tokens * 4]

#########################
# FASTAPI APP LOGIC
#########################
app = FastAPI()

@app.post("/batch_autosummarize_pdf/")
async def batch_autosummarize_pdf(
    file: UploadFile = File(...)
):
    raw = await file.read()
    pages = extract_pdf_text(raw)
    chunks = chunk_with_langchain(pages)
    # Async batch call
    results = await asyncio.gather(*(async_gpt_summarize(chunk) for chunk in chunks))
    summaries = []
    for res, chunk in zip(results, chunks):
        try:
            # If the LLM already responded as JSON:
            import json
            parsed = json.loads(res)
        except Exception:
            # Fallback if not, wrap in simple dict
            parsed = {
                "section_title": None,
                "summary": res,
                "confidence": None,
                "chunk_snippet": chunk[:120]
            }
        summaries.append(parsed)
    return {"ok": True, "summaries": summaries}


@app.post("/batch_autosummarize_texts/")
async def batch_autosummarize_texts(payload: Dict[str, Any]):
    chunks = chunk_with_langchain(payload['texts'])
    results = await asyncio.gather(*(async_gpt_summarize(chunk) for chunk in chunks))
    summaries = []
    for res, chunk in zip(results, chunks):
        try:
            import json
            parsed = json.loads(res)
        except:
            parsed = {"section_title": None, "summary": res, "confidence": None}
        summaries.append(parsed)
    return {"ok": True, "summaries": summaries}
