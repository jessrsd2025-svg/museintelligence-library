A. **Python FastAPI Code**


```
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
from typing import List, Optional

app = FastAPI(title="Disinformation Tactics API")

class Tactic(BaseModel):
    id: str
    name: str
    description: str
    examples: List[str]
    countermeasures: List[str]

# Simple in-memory DB (replace with real DB in prod)
tactic_db = {}

@app.post("/tactics/", response_model=Tactic)
def add_tactic(tactic: Tactic):
    if tactic.id in tactic_db:
        raise HTTPException(status_code=400, detail="Tactic ID exists!")
    tactic_db[tactic.id] = tactic
    return tactic

@app.get("/tactics/", response_model=List[Tactic])
def list_tactics():
    return list(tactic_db.values())

@app.get("/tactics/{tactic_id}", response_model=Tactic)
def get_tactic(tactic_id: str):
    if tactic_id not in tactic_db:
        raise HTTPException(status_code=404, detail="Not found")
    return tactic_db[tactic_id]

@app.delete("/tactics/{tactic_id}")
def delete_tactic(tactic_id: str):
    if tactic_id not in tactic_db:
        raise HTTPException(status_code=404, detail="Not found")
    del tactic_db[tactic_id]
    return {"detail": "Deleted"}

# Example curl:
# curl -X POST "http://localhost:8000/tactics/" -H "Content-Type: application/json" -d '{"id":"deepfake","name":"Deepfakes & Synthetic Media","description":"AI-generated/manipulated media for disinfo","examples":["Spamouflage Dragon videos"],"countermeasures":["Reverse image search","Incident response"]}'
```


**To run it:** Save as `main.py`, then run `uvicorn main:app --reload`.
