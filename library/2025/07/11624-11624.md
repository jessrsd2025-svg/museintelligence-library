# 👇 “Definition of a fentanyl-related substance under the HALT Act” using RetrievalQA


```
from langchain.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.vectorstores.faiss import FAISS
from langchain_openai import OpenAIEmbeddings, ChatOpenAI
from langchain.chains import RetrievalQA
import os

# 1. Chunk PDF into Docs
def pdf_to_docs(pdf_path):
    loader = PyPDFLoader(pdf_path)
    pages = loader.load_and_split()
    splitter = RecursiveCharacterTextSplitter(chunk_size=1024, chunk_overlap=100)
    return splitter.split_documents(pages)

# 2. Embed and Create Retriever (uses FAISS for speed; Chroma is also drop-in)
def docs_to_retriever(docs):
    embeddings = OpenAIEmbeddings(openai_api_key=os.environ["OPENAI_API_KEY"])
    vectorstore = FAISS.from_documents(docs, embeddings)
    return vectorstore.as_retriever(search_type="similarity", search_kwargs={"k": 4})

# 3. Compose RetrievalQA chain
def setup_qa_chain(retriever):
    llm = ChatOpenAI(model="gpt-4o", openai_api_key=os.environ["OPENAI_API_KEY"])
    return RetrievalQA.from_chain_type(
        llm=llm,
        retriever=retriever,
        return_source_documents=True,
    )

if __name__ == "__main__":
    # --- Step 1: Ingest HALT Act PDF ---
    docs = pdf_to_docs("HALT_Fentanyl_Act.pdf")
    # --- Step 2: Create Retriever ---
    retriever = docs_to_retriever(docs)
    # --- Step 3: Setup QA chain ---
    qa_chain = setup_qa_chain(retriever)
    
    # --- Step 4: Ask and get cited answer ---
    query = "What is the definition of a fentanyl-related substance under the HALT Act?"
    result = qa_chain.invoke({"query": query})

    print("HALT Act Answer:\n", result["result"])
    print("\nSource Excerpts:")
    for doc in result["source_documents"]:
        print("-", doc.page_content[:400], "\n[...]\n")
```

```
from langchain.schema.document import Document

def make_documents(crisis_objs):
    docs = []
    for crisis in crisis_objs:
        # Combine key fields to maximize semantic signal
        content = (
            crisis.summary + " "
            + " ".join(crisis.references) + " "
            + " ".join(crisis.metrics.lessons_tags)

        docs.append(
            Document(

                    "impact_score": crisis.metrics.impact_score,
                    "fatality_score": crisis.metrics.fatality_score,
                    "reform_lag": crisis.metrics.reform_lag,
                    "location_tags": getattr(crisis, 'location_tags', []),
                }
            )
        )
    return docs

docs = make_documents([nitazenes, ebola, cholera])  # Your instantiated objects
```

## 3. Semantic Retrieval + Q&A


```
from langchain.chains import RetrievalQA
from langchain_openai import ChatOpenAI

llm = ChatOpenAI(model="gpt-4o", temperature=0)

qa_chain = RetrievalQA.from_chain_type(
    llm=llm,
    retriever=retriever,
    return_source_documents=True,  # For audit logs or dashboard links
)

query = "Which crisis showed the slowest community trust recovery after reforms?"
result = qa_chain.invoke({"query": query})
print("Best Answer: ", result["result"])
print("Top Source Doc(s):", [doc.page_content[:200] for doc in result["source_documents"]])
```
